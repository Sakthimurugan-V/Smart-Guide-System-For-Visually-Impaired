MAX_AREA = 5000

MIN_CONFIDENCE = 0.55


# How often the same signal can be announced
ANNOUNCEMENT_DELAY = 3.0

# ============================================================
# TRAFFIC LIGHT COLOR RANGES
# HSV COLOR SPACE
# ============================================================

# RED
RED_LOWER1 = np.array(
    [0, 100, 100]
)

RED_UPPER1 = np.array(
    [10, 255, 255]
)

RED_LOWER2 = np.array(
    [170, 100, 100]
)

RED_UPPER2 = np.array(
    [179, 255, 255]
)


# YELLOW
YELLOW_LOWER = np.array(
    [18, 100, 100]
)

YELLOW_UPPER = np.array(
    [40, 255, 255]
)


# GREEN
GREEN_LOWER = np.array(
    [40, 70, 60]
)

GREEN_UPPER = np.array(
    [90, 255, 255]
)


# ============================================================
# CREATE COLOR MASK
# ============================================================

def create_color_mask(
    hsv,
    color
):

    if color == "RED":

        mask1 = cv2.inRange(
            hsv,
            RED_LOWER1,
            RED_UPPER1
        )

        mask2 = cv2.inRange(
            hsv,
            RED_LOWER2,
            RED_UPPER2
        )

        mask = cv2.bitwise_or(
            mask1,
            mask2
        )

    elif color == "YELLOW":

        mask = cv2.inRange(
            hsv,
            YELLOW_LOWER,
            YELLOW_UPPER
        )

    elif color == "GREEN":

        mask = cv2.inRange(
            hsv,
            GREEN_LOWER,
            GREEN_UPPER
        )

    else:

        mask = np.zeros(
            hsv.shape[:2],
            dtype=np.uint8
        )


    # Remove small noise
    kernel = np.ones(
        (5, 5),
        np.uint8
    )


    mask = cv2.morphologyEx(
        mask,
        cv2.MORPH_OPEN,
        kernel
    )


    mask = cv2.morphologyEx(
        mask,
        cv2.MORPH_CLOSE,
        kernel
    )


    return mask


# ============================================================
# SHAPE FILTERING
# ============================================================

def check_shape(
    contour
):

    area = cv2.contourArea(
        contour
    )


    if area < MIN_AREA:
        return False, 0


    if area > MAX_AREA:
        return False, 0


    perimeter = cv2.arcLength(
        contour,
        True
    )


    if perimeter == 0:
        return False, 0


    # Circularity:
    #
    #     4 * pi * Area
    #     ----------------
    #       Perimeter²
    #
    # Circle ≈ 1.0

    circularity = (
        4 *
        np.pi *
        area
        /
        (perimeter * perimeter)
    )


    if circularity < MIN_CIRCULARITY:

        return False, circularity


    # Bounding rectangle

    x, y, w, h = cv2.boundingRect(
        contour
    )


    aspect_ratio = (
        float(w) /
        float(h)
    )


    # Traffic light bulbs are approximately circular

    if aspect_ratio < 0.55:

        return False, circularity


    if aspect_ratio > 1.80:

        return False, circularity


    return True, circularity


# ============================================================
# DETECT ONE COLOR
# ============================================================

def detect_color(
    hsv,
    color
):

    mask = create_color_mask(
        hsv,
        color
    )


    contours, _ = cv2.findContours(
        mask,
        cv2.RETR_EXTERNAL,
        cv2.CHAIN_APPROX_SIMPLE
    )


    best_detection = None

    best_score = 0


    for contour in contours:

        valid, circularity = check_shape(
            contour
        )


        if not valid:
            continue


        area = cv2.contourArea(
            contour
        )


        x, y, w, h = cv2.boundingRect(
            contour
        )


        # Calculate center

        M = cv2.moments(
            contour
        )


        if M["m00"] == 0:
            continue


        cx = int(
            M["m10"] /
            M["m00"]
        )


        cy = int(
            M["m01"] /
            M["m00"]
        )


        # ----------------------------------------------------
        # SCORE
        # ----------------------------------------------------

        # Larger area = better

        area_score = min(
            area / 1000.0,
            1.0
        )


        # Circularity = better shape

        shape_score = min(
            circularity,
            1.0
        )


        score = (
            0.45 *
            area_score
            +
            0.55 *
            shape_score
        )


        if score > best_score:

            best_score = score

            best_detection = {

                "color": color,

                "x": x,

                "y": y,

                "w": w,

                "h": h,

                "cx": cx,

                "cy": cy,

                "area": area,

                "circularity": circularity,

                "score": score

            }


    return (
        best_detection,
        mask
    )


# ============================================================
# DETECT TRAFFIC LIGHT
# ============================================================

def detect_traffic_light(
    frame
):

    # --------------------------------------------------------
    # ROI
    # --------------------------------------------------------

    roi = frame[
        ROI_Y1:ROI_Y2,
        ROI_X1:ROI_X2
    ]


    # --------------------------------------------------------
    # HSV
    # --------------------------------------------------------

    hsv = cv2.cvtColor(
        roi,
        cv2.COLOR_RGB2HSV
    )


    detections = []


    # --------------------------------------------------------
    # RED
    # --------------------------------------------------------

    red_detection, red_mask = detect_color(
        hsv,
        "RED"
    )


    if red_detection is not None:

        detections.append(
            red_detection
        )


    # --------------------------------------------------------
    # YELLOW
    # --------------------------------------------------------

    yellow_detection, yellow_mask = detect_color(
        hsv,
        "YELLOW"
    )


    if yellow_detection is not None:

        detections.append(
            yellow_detection
        )


    # --------------------------------------------------------
    # GREEN
    # --------------------------------------------------------

    green_detection, green_mask = detect_color(
        hsv,
        "GREEN"
    )


    if green_detection is not None:

        detections.append(
            green_detection
        )


    # --------------------------------------------------------
    # NO DETECTION
    # --------------------------------------------------------

    if len(detections) == 0:

        return None


    # --------------------------------------------------------
    # BEST DETECTION
    # --------------------------------------------------------

    detections.sort(
        key=lambda x: x["score"],
        reverse=True
    )


    best = detections[0]


    # Convert ROI coordinates
    # back to full-frame coordinates

    best["x"] += ROI_X1

    best["y"] += ROI_Y1

    best["cx"] += ROI_X1

    best["cy"] += ROI_Y1


    return best


# ============================================================
# STABLE DETECTION
# ============================================================

def stable_signal(
    signal
):

    global signal_history


    signal_history.append(
        signal
    )


    # Keep last 7 detections

    if len(signal_history) > 7:

        signal_history.pop(
            0
        )


    if len(signal_history) < 5:

        return None


    # Count signals

    counts = {}


    for item in signal_history:

        if item not in counts:

            counts[item] = 0

        counts[item] += 1


    # Find most common

    best_signal = max(
        counts,
        key=counts.get
    )


    best_count = counts[
        best_signal
    ]


    # At least 4 out of last 7
    # frames must agree

    if best_count >= 4:

        return best_signal


    return None


# ============================================================
# ANNOUNCE SIGNAL
# ============================================================

def announce_signal(
    signal
):

    global last_signal
    global last_announcement_time


    current_time = time.time()


    # Don't repeat too frequently

    if (
        signal ==
        last_signal
        and
        current_time -
        last_announcement_time
        <
        ANNOUNCEMENT_DELAY
    ):

        return


    if (
        current_time -
        last_announcement_time
        <
        ANNOUNCEMENT_DELAY
    ):

        return


    # --------------------------------------------------------
    # RED
    # --------------------------------------------------------

    if signal == "RED":

        speak(
            "Red signal. Stop."
        )


    # --------------------------------------------------------
    # YELLOW
    # --------------------------------------------------------

    elif signal == "YELLOW":

        speak(
            "Yellow signal. Wait."
        )


    # --------------------------------------------------------
    # GREEN
    # --------------------------------------------------------

    elif signal == "GREEN":

        speak(
            "Green signal. Go."
        )


    last_signal =
        signal


    last_announcement_time =
        current_time
